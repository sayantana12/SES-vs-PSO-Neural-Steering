import numpy as np
import matplotlib.pyplot as plt

def run_and_log_final_errors(use_pso, num_robots=6, frames=60, seed=123):
    # --- Copied from AnimatedPathPlanner, but logs final distance per robot ---
    class BarChartPlanner:
        def __init__(self):
            self.num_robots = num_robots
            self.workspace_size = 40
            np.random.seed(seed)
            self.robot_positions = np.random.uniform(5, 35, (num_robots, 2))
            self.robot_goals = np.random.uniform(5, 35, (num_robots, 2))
            self.max_velocity = 3.0
            self.dt = 1.5
            self.influence_distance = 8.0
            self.ksi_att = 0.5
            self.ksi_rep = 0.3
            self.input_size = 3
            self.hidden_size = 10
            self.output_size = 1
            self.obstacles = [
                {'type': 'circle', 'center': (20, 20), 'radius': 4},
                {'type': 'rect', 'bottom_left': (10, 5), 'width': 4, 'height': 8}
            ]
            self.initialize_ann_structure()
            self.train_network()
        def initialize_ann_structure(self):
            self.w1_size = self.input_size * self.hidden_size
            self.b1_size = self.hidden_size
            self.w2_size = self.hidden_size * self.output_size
            self.b2_size = self.output_size
            self.vector_size = self.w1_size + self.b1_size + self.w2_size + self.b2_size
        def sigmoid(self, x): return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
        def tanh(self, x): return np.tanh(np.clip(x, -500, 500))
        def angle_diff(self, a, b): return np.arctan2(np.sin(a - b), np.cos(a - b))
        def ann_forward(self, X, W1, b1, W2, b2):
            hidden = self.sigmoid(np.dot(X, W1) + b1)
            output = self.tanh(np.dot(hidden, W2) + b2) * np.pi
            return output
        def decode_ann(self, particle):
            idx = 0
            W1 = particle[idx:idx+self.w1_size].reshape(self.input_size, self.hidden_size)
            idx += self.w1_size
            b1 = particle[idx:idx+self.b1_size]
            idx += self.b1_size
            W2 = particle[idx:idx+self.w2_size].reshape(self.hidden_size, self.output_size)
            idx += self.w2_size
            b2 = particle[idx:idx+self.b2_size]
            return W1, b1, W2, b2
        def generate_training_data(self):
            num_samples = 2000
            self.X_train = np.zeros((num_samples, 3))
            self.y_train = np.zeros(num_samples)
            for i in range(num_samples):
                goal_angle = np.random.uniform(-np.pi, np.pi)
                crit_angle = np.random.uniform(-np.pi, np.pi)
                crit_dist = np.random.uniform(1.0, self.influence_distance+2.0)
                F_att = self.ksi_att * crit_dist
                F_rep = self.ksi_rep * ((1/crit_dist - 1/self.influence_distance) / (crit_dist**2)) if crit_dist < self.influence_distance else 0
                deviation = np.arctan2(F_rep * np.sin(crit_angle) + F_att * np.sin(goal_angle),
                                       F_rep * np.cos(crit_angle) + F_att * np.cos(goal_angle))
                self.X_train[i] = [goal_angle, crit_angle, crit_dist]
                self.y_train[i] = deviation
        def diversity_penalty(self, population):
            if len(population) <= 1: return 0.0
            centroid = np.mean(population, axis=0)
            diversity = np.mean(np.linalg.norm(population - centroid, axis=1))
            return 1.0 / (diversity + 1e-6)
        def train_network(self):
            self.generate_training_data()
            population = np.random.uniform(-1, 1, (30, self.vector_size))
            velocities = np.zeros_like(population)
            p_best = population.copy()
            p_best_scores = np.full(30, np.inf)
            g_best = None
            g_best_score = np.inf
            if not use_pso:
                for gen in range(70):
                    fitness_scores = []
                    for i in range(30):
                        W1, b1, W2, b2 = self.decode_ann(population[i])
                        y_pred = self.ann_forward(self.X_train, W1, b1, W2, b2).flatten()
                        mse = np.mean(self.angle_diff(self.y_train, y_pred)**2)
                        fitness_scores.append(mse)
                    fitness_scores = np.array(fitness_scores)
                    penalty = self.diversity_penalty(population)
                    total_fitness = fitness_scores + 0.1 * penalty
                    elite_idx = np.argmin(total_fitness)
                    elite = population[elite_idx]
                    new_population = [elite.copy()]
                    while len(new_population) < 30:
                        mutation = np.random.normal(0, 0.1, size=self.vector_size)
                        child = elite + mutation
                        new_population.append(np.clip(child, -5, 5))
                    population = np.array(new_population)
                self.best_weights = self.decode_ann(elite)
            else:
                w_init, w_final = 0.9, 0.4
                c1 = c2 = 1.49445
                v_clamp = 1.0
                for gen in range(70):
                    w = w_init - (w_init - w_final) * (gen / 70)
                    for i in range(30):
                        W1, b1, W2, b2 = self.decode_ann(population[i])
                        y_pred = self.ann_forward(self.X_train, W1, b1, W2, b2).flatten()
                        mse = np.mean(self.angle_diff(self.y_train, y_pred) ** 2)
                        penalty = self.diversity_penalty(population)
                        fitness = mse + 0.05 * penalty
                        if fitness < p_best_scores[i]:
                            p_best[i] = population[i]
                            p_best_scores[i] = fitness
                        if fitness < g_best_score:
                            g_best = population[i]
                            g_best_score = fitness
                    for i in range(30):
                        r1, r2 = np.random.rand(self.vector_size), np.random.rand(self.vector_size)
                        velocities[i] = w * velocities[i] \
                                    + c1 * r1 * (p_best[i] - population[i]) \
                                    + c2 * r2 * (g_best - population[i])
                        velocities[i] = np.clip(velocities[i], -v_clamp, v_clamp)
                        population[i] = np.clip(population[i] + velocities[i], -5, 5)
                    if gen % 20 == 0:
                        idxs = np.random.choice(30, 2, replace=False)
                        for idx in idxs:
                            mutation = np.random.normal(0, 0.1, size=self.vector_size)
                            population[idx] = np.clip(population[idx] + mutation, -5, 5)
                self.best_weights = self.decode_ann(g_best)
        def is_collision(self, pos):
            for obs in self.obstacles:
                if obs['type'] == 'circle':
                    cx, cy = obs['center']
                    if np.linalg.norm(np.array(pos) - np.array([cx, cy])) <= obs['radius']+1.0:
                        return True
                elif obs['type'] == 'rect':
                    x, y = pos
                    x0, y0 = obs['bottom_left']
                    if x0 <= x <= x0+obs['width'] and y0 <= y <= y0+obs['height']:
                        return True
            return False
        def get_final_distances(self):
            positions = np.copy(self.robot_positions)
            reached = [False] * self.num_robots
            for frame in range(frames):
                W1, b1, W2, b2 = self.best_weights
                for i in range(self.num_robots):
                    if reached[i]: continue
                    goal_dir = self.robot_goals[i] - positions[i]
                    goal_dist = np.linalg.norm(goal_dir)
                    if goal_dist < 1.0:
                        reached[i] = True
                        continue
                    goal_dir /= goal_dist + 1e-6
                    angle_to_goal = np.arctan2(goal_dir[1], goal_dir[0])
                    deviation_angle = 0
                    for j in range(self.num_robots):
                        if i != j:
                            dist = np.linalg.norm(positions[i] - positions[j])
                            if dist < 2 * self.influence_distance:
                                critical_angle = np.arctan2(
                                    positions[j][1] - positions[i][1],
                                    positions[j][0] - positions[i][0]
                                )
                                input_vec = np.array([[angle_to_goal, critical_angle, dist]])
                                deviation_angle = self.ann_forward(input_vec, W1, b1, W2, b2)[0, 0]
                                break
                    rot = np.array([[np.cos(deviation_angle), -np.sin(deviation_angle)],[np.sin(deviation_angle), np.cos(deviation_angle)]])
                    new_dir = np.dot(rot, goal_dir)
                    velocity = new_dir * self.max_velocity
                    next_pos = positions[i] + velocity * self.dt
                    # Check for obstacle
                    if self.is_collision(next_pos):
                        new_dir = np.dot(rot, goal_dir + np.random.uniform(-0.5, 0.5, size=2))
                        next_pos = positions[i] + new_dir * self.max_velocity * self.dt
                    if not self.is_collision(next_pos): positions[i] = next_pos
            # Now compute final distances
            dists = [np.linalg.norm(positions[i] - self.robot_goals[i]) for i in range(self.num_robots)]
            return dists
    return BarChartPlanner().get_final_distances()

robots = 6
ses_distances = run_and_log_final_errors(use_pso=False, num_robots=robots)
pso_distances = run_and_log_final_errors(use_pso=True, num_robots=robots)

labels = [f"Robot {i+1}" for i in range(robots)]
x = np.arange(robots)
width = 0.38

plt.figure(figsize=(9,6))
bars1 = plt.bar(x - width/2, ses_distances, width=width, label="SES", color='skyblue')
bars2 = plt.bar(x + width/2, pso_distances, width=width, label="PSO", color='orange')

plt.ylabel('Final Distance to Goal (lower is better)')
plt.xlabel('Robot Number')
plt.title('Figure C: Per-Robot Final Distance to Goal – SES vs PSO')
plt.xticks(x, labels)
plt.legend()
plt.ylim(0, max(max(ses_distances), max(pso_distances)) * 1.1)
for bar, val in zip(bars1, ses_distances):
    plt.text(bar.get_x() + bar.get_width()/2, bar.get_height()+0.1, f'{val:.2f}', ha='center', color='blue')
for bar, val in zip(bars2, pso_distances):
    plt.text(bar.get_x() + bar.get_width()/2, bar.get_height()+0.1, f'{val:.2f}', ha='center', color='darkorange')
plt.tight_layout()
plt.show()
