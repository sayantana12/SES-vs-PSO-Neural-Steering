# Multi-Robot Path Planning using Swarm Evolutionary Strategy (SES) and Particle swarm optimization(pso) with Diversity Penalty and Excel Export

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation
from sklearn.metrics import mean_squared_error, mean_absolute_error
from tqdm import trange
import time
import pandas as pd

class MultiRobotPathPlanner:
    def __init__(self, num_robots=8, workspace_size=40, max_velocity=5.0, max_acceleration=0.05,
                 dt=2.0, influence_distance=8.0, population_size=30, generations=100):

        self.num_robots = num_robots
        self.workspace_size = workspace_size
        self.max_velocity = max_velocity
        self.max_acceleration = max_acceleration
        self.dt = dt
        self.influence_distance = influence_distance
        self.population_size = population_size
        self.generations = generations
        self.robot_deviation_angles = np.zeros(self.num_robots)
        self.ksi_att = 0.5
        self.ksi_rep = 0.3

        np.random.seed(42)
        self.robot_positions = np.random.uniform(5, workspace_size-5, (num_robots, 2))
        self.robot_goals = np.random.uniform(5, workspace_size-5, (num_robots, 2))
        self.robot_velocities = np.zeros((num_robots, 2))

        self.input_size = 3
        self.hidden_size = 10
        self.output_size = 1

        self.travelling_times = np.zeros(num_robots)
        self.cooperation_counts = np.zeros(num_robots)
        self.deviation_logs = []
        self.att_history = []
        self.paths = [[] for _ in range(self.num_robots)]

        self.initialize_ann_structure()
        self.generate_training_data()
        self.train_ann_with_ses()
        #self.train_ann_with_pso()

    def sigmoid(self, x):
        x = np.clip(x, -500, 500)
        return 1 / (1 + np.exp(-x))

    def tanh(self, x):
        x = np.clip(x, -500, 500)
        return np.tanh(x)

    def angle_diff(self, a, b):
        return np.arctan2(np.sin(a - b), np.cos(a - b))

    def initialize_ann_structure(self):
        self.w1_size = self.input_size * self.hidden_size
        self.b1_size = self.hidden_size
        self.w2_size = self.hidden_size * self.output_size
        self.b2_size = self.output_size
        self.vector_size = self.w1_size + self.b1_size + self.w2_size + self.b2_size

    def decode_ann(self, particle):
        idx = 0
        W1 = particle[idx:idx + self.w1_size].reshape(self.input_size, self.hidden_size)
        idx += self.w1_size
        b1 = particle[idx:idx + self.b1_size]
        idx += self.b1_size
        W2 = particle[idx:idx + self.w2_size].reshape(self.hidden_size, self.output_size)
        idx += self.w2_size
        b2 = particle[idx:idx + self.b2_size]
        return W1, b1, W2, b2

    def ann_forward(self, X, W1, b1, W2, b2):
        hidden = self.sigmoid(np.dot(X, W1) + b1)
        output = self.tanh(np.dot(hidden, W2) + b2) * np.pi
        return output

    def generate_training_data(self):
        num_samples = 2000
        self.X_train = np.zeros((num_samples, 3))
        self.y_train = np.zeros(num_samples)

        for i in range(num_samples):
            goal_angle = np.random.uniform(-np.pi, np.pi)
            crit_angle = np.random.uniform(-np.pi, np.pi)
            crit_dist = np.random.uniform(1.0, self.influence_distance + 2.0)
            F_att = self.ksi_att * crit_dist
            if crit_dist < self.influence_distance:
                F_rep = self.ksi_rep * ((1/crit_dist - 1/self.influence_distance) / (crit_dist**2))
            else:
                F_rep = 0
            deviation = np.arctan2(F_rep * np.sin(crit_angle) + F_att * np.sin(goal_angle),
                                   F_rep * np.cos(crit_angle) + F_att * np.cos(goal_angle))
            self.X_train[i] = [goal_angle, crit_angle, crit_dist]
            self.y_train[i] = deviation

    def diversity_penalty(self, population):
        if len(population) <= 1:
            return 0.0
        centroid = np.mean(population, axis=0)
        diversity = np.mean(np.linalg.norm(population - centroid, axis=1))
        return 1.0 / (diversity + 1e-6)

    def train_ann_with_ses(self):
        population = np.random.uniform(-1, 1, (self.population_size, self.vector_size))
        fitness_history = []

        for generation in trange(self.generations, desc="Training with SES"):
            fitness_scores = []

            for i in range(self.population_size):
                W1, b1, W2, b2 = self.decode_ann(population[i])
                y_pred = self.ann_forward(self.X_train, W1, b1, W2, b2).flatten()
                mse = np.mean(self.angle_diff(self.y_train, y_pred)**2)
                fitness_scores.append(mse)

            fitness_scores = np.array(fitness_scores)
            penalty = self.diversity_penalty(population)
            total_fitness = fitness_scores + 0.1 * penalty

            elite_idx = np.argmin(total_fitness)
            elite = population[elite_idx]
            new_population = [elite.copy()]

            while len(new_population) < self.population_size:
                parent = elite.copy()
                mutation = np.random.normal(0, 0.1, size=self.vector_size)
                child = parent + mutation
                new_population.append(np.clip(child, -5, 5))

            population = np.array(new_population)
            fitness_history.append(np.min(fitness_scores))

        self.best_weights = self.decode_ann(population[np.argmin(fitness_scores)])

        self.plot_training_results(fitness_history)
        self.animate_robot_movement()
    def train_ann_with_pso(self):
        population = np.random.uniform(-1, 1, (self.population_size, self.vector_size))
        velocities = np.zeros_like(population)

        p_best = population.copy()
        p_best_scores = np.full(self.population_size, np.inf)

        g_best = None
        g_best_score = np.inf

        fitness_history = []

        # PSO Hyperparameters
        w = 0.729
        c1 = 1.49445
        c2 = 1.49445

        for gen in trange(self.generations, desc="Training with PSO"):
            for i in range(self.population_size):
                W1, b1, W2, b2 = self.decode_ann(population[i])
                y_pred = self.ann_forward(self.X_train, W1, b1, W2, b2).flatten()
                mse = np.mean(self.angle_diff(self.y_train, y_pred) ** 2)

                if mse < p_best_scores[i]:
                    p_best[i] = population[i]
                    p_best_scores[i] = mse

                if mse < g_best_score:
                    g_best = population[i]
                    g_best_score = mse

            for i in range(self.population_size):
                r1 = np.random.rand(self.vector_size)
                r2 = np.random.rand(self.vector_size)
                velocities[i] = (
                    w * velocities[i]
                    + c1 * r1 * (p_best[i] - population[i])
                    + c2 * r2 * (g_best - population[i])
                )
                population[i] += velocities[i]
                population[i] = np.clip(population[i], -5, 5)

            fitness_history.append(g_best_score)

        self.best_weights = self.decode_ann(g_best)
        self.plot_training_results(fitness_history)
        self.animate_robot_movement()




    def plot_training_results(self, fitness_history):
        y_pred = self.ann_forward(self.X_train, *self.best_weights).flatten()
        errors = self.angle_diff(self.y_train, y_pred)

        plt.figure()
        plt.hist(errors, bins=50, alpha=0.7, color='skyblue')
        plt.title("Prediction Error Histogram")
        plt.xlabel("Angle Error (rad)")
        plt.ylabel("Frequency")
        plt.grid()
        plt.show()

        plt.figure()
        plt.plot(fitness_history, label='Best MSE')
        plt.title("Fitness Progress during SES Training")
        plt.xlabel("Generation")
        plt.ylabel("MSE")
        plt.grid()
        plt.legend()
        plt.show()

        if hasattr(self, 'att_history') and len(self.att_history) > 0:
            plt.figure()
            plt.plot(self.att_history, color='green', label='ATT Over Time')
            plt.title("Average Travelling Time Over Time")
            plt.xlabel("Timestep")
            plt.ylabel("Average Time (s)")
            plt.grid(True)
            plt.legend()
            plt.show()

        if hasattr(self, 'paths'):
            plt.figure(figsize=(8, 8))
            for i in range(self.num_robots):
                path = np.array(self.paths[i])
                if len(path) > 0:
                    plt.plot(path[:, 0], path[:, 1], label=f'Robot {i+1}')
                plt.plot(self.robot_goals[i, 0], self.robot_goals[i, 1], 'rx')
            plt.title("Robot Travel Paths")
            plt.xlabel("X")
            plt.ylabel("Y")
            plt.grid(True)
            plt.legend()
            plt.show()

    def animate_robot_movement(self):
        fig, ax = plt.subplots(figsize=(8, 8))
        ax.set_xlim(0, self.workspace_size)
        ax.set_ylim(0, self.workspace_size)
        ax.set_title("Robot Movement Animation")

        robots_plot = [ax.plot([], [], 'o')[0] for _ in range(self.num_robots)]
        goals_plot = [ax.plot(self.robot_goals[i, 0], self.robot_goals[i, 1], 'x', markersize=10)[0] for i in range(self.num_robots)]
        path_lines = [ax.plot([], [], '--')[0] for _ in range(self.num_robots)]

        start_time = time.time()
        reached = [False] * self.num_robots
        #dev_angle = self.ann_forward(input_vec, W1, b1, W2, b2)[0, 0]
        #self.robot_deviation_angles[i] += abs(dev_angle)  # Accumulate absolute deviation

        def update(frame):
            W1, b1, W2, b2 = self.best_weights
            for i in range(self.num_robots):
                if reached[i]:
                    continue
                goal_dir = self.robot_goals[i] - self.robot_positions[i]
                goal_dist = np.linalg.norm(goal_dir)
                if goal_dist < 1.0:
                    self.travelling_times[i] = time.time() - start_time
                    reached[i] = True
                    continue
                goal_dir /= goal_dist + 1e-6
                angle_to_goal = np.arctan2(goal_dir[1], goal_dir[0])

                for j in range(self.num_robots):
                    if i != j:
                        dist = np.linalg.norm(self.robot_positions[i] - self.robot_positions[j])
                        if dist < 2 * self.influence_distance:
                            critical_angle = np.arctan2(
                                self.robot_positions[j][1] - self.robot_positions[i][1],
                                self.robot_positions[j][0] - self.robot_positions[i][0]
                            )
                            input_vec = np.array([[angle_to_goal, critical_angle, dist]])   #input parameter
                            dev_angle = self.ann_forward(input_vec, W1, b1, W2, b2)[0, 0]
                            self.cooperation_counts[i] += 1
                            self.robot_deviation_angles[i] += abs(dev_angle)
                            self.deviation_logs.append({
                                             'Robot': i + 1,
                                             'Goal Angle': angle_to_goal,
                                             'Crit Angle': critical_angle,
                                             'Crit Dist': dist,
                                             'Deviation': dev_angle
                                                      })
                            rot = np.array([[np.cos(dev_angle), -np.sin(dev_angle)],
                                            [np.sin(dev_angle),  np.cos(dev_angle)]])
                            new_dir = np.dot(rot, goal_dir)
                            velocity = new_dir * self.max_velocity
                            self.robot_positions[i] += velocity * self.dt
                            break

                self.paths[i].append(self.robot_positions[i].copy())
                robots_plot[i].set_data([self.robot_positions[i][0]], [self.robot_positions[i][1]])
                xs, ys = zip(*self.paths[i])
                path_lines[i].set_data(xs, ys)

            att = np.mean(self.travelling_times[self.travelling_times > 0]) if np.any(self.travelling_times > 0) else 0
            self.att_history.append(att)

            return robots_plot + path_lines + goals_plot

        ani = FuncAnimation(fig, update, frames=200, interval=300, blit=True, repeat=False)
        plt.legend()
        plt.grid(True)
        plt.show()

        self.export_to_excel()

    def export_to_excel(self):
        df_deviation = pd.DataFrame(self.deviation_logs)
        df_stats = pd.DataFrame({
            'Robot': np.arange(1, self.num_robots + 1),
            'Travel Time (s)': self.travelling_times,
            'Cooperation Count': self.cooperation_counts
        })
        df_att = pd.DataFrame({'Time Step': np.arange(len(self.att_history)), 'ATT': self.att_history})

        with pd.ExcelWriter("robot_coordination_results.xlsx") as writer:
            df_deviation.to_excel(writer, sheet_name="Deviations", index=False)
            df_stats.to_excel(writer, sheet_name="Stats", index=False)
            df_att.to_excel(writer, sheet_name="ATT_Over_Time", index=False)
        print("Results saved to 'robot_coordination_results.xlsx'")

    def result_summary(self):
        W1, b1, W2, b2 = self.best_weights
        y_pred = self.ann_forward(self.X_train, W1, b1, W2, b2).flatten()

        print("\nRESULT SUMMARY")
        print("="*60)
        print("First 10 predictions vs actual values:")
        for i in range(10):
            print(f"Sample {i+1:2d}: Predicted = {y_pred[i]:.3f} rad, Actual = {self.y_train[i]:.3f} rad, Error = {self.angle_diff(self.y_train[i], y_pred[i]):.3f}")

        mse = np.mean((self.angle_diff(self.y_train, y_pred))**2)
        mae = mean_absolute_error(self.y_train, y_pred)
        within_half_rad = np.mean(np.abs(self.angle_diff(self.y_train, y_pred)) < 0.5)

        print(f"\nMean Squared Error: {mse:.6f}")
        print(f"Mean Absolute Error: {mae:.6f}")
        print(f"% predictions within ±0.5 rad: {within_half_rad * 100:.2f}%")
        print("\nRobot-wise Total Deviation Summary (in radians):")
        for i in range(self.num_robots):
                  total_dev = self.robot_deviation_angles[i]
                  steps = len(self.paths[i]) if len(self.paths[i]) > 0 else 1
                  print(f"Robot {i+1:2d}: Total Deviation = {total_dev:.3f} rad, "f"Avg per Step ≈ {total_dev/steps:.3f} rad (over {steps} steps)")
